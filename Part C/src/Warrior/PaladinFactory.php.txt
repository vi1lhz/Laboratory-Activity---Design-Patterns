<?php

namespace App\Warrior;

use App\Contracts\CharacterFactory;
use App\Contracts\Weapon;
use App\Contracts\Armor;
use App\Contracts\Ability;

class PaladinFactory implements CharacterFactory
{
    public function createWeapon(): Weapon
    {
        return new Warhammer();
    }

    public function createArmor(): Armor
    {
        return new SacredArmor();
    }

    public function createAbility(): Ability
    {
        return new HolySmite();
    }
}
