class CallingCardPrototype
{
    private string $firstName;
    private string $lastName;
    private string $businessName;
    private string $position;
    private string $street;
    private string $city;
    private string $website;

    public function __construct(
        string $businessName = 'College of Computing Studies',
        string $position = 'BSIT Student',
        string $street = 'AUF CCS Building',
        string $city = 'Angeles City',
        string $website = 'www.auf.edu.ph'
    ) {
        $this-businessName = $businessName;
        $this-position = $position;
        $this-street = $street;
        $this-city = $city;
        $this-website = $website;
    }

    public function setStudentDetails(string $firstName, string $lastName) self
    {
        $this-firstName = $firstName;
        $this-lastName = $lastName;
        return $this;
    }

    public function __clone() {}

    public function render() GdImage
    {
        $width = 1000;
        $height = 600;

        $image = imagecreatetruecolor($width, $height);

        $background = imagecolorallocate($image, 245, 247, 250);
        $white      = imagecolorallocate($image, 255, 255, 255);
        $black      = imagecolorallocate($image, 30, 30, 30);
        $gray       = imagecolorallocate($image, 100, 100, 100);
        $blue       = imagecolorallocate($image, 40, 100, 200);

        imagefill($image, 0, 0, $background);

        imagefilledrectangle($image, 50, 50, 950, 550, $white);
        imagefilledrectangle($image, 50, 50, 75, 550, $blue);

        $name = {$this-firstName} {$this-lastName};
        $email = strtolower({$this-lastName}.{$this-firstName}@auf.edu.ph);
        $phone = sprintf('+1 (555) %03d-%04d', rand(100, 999), rand(1000, 9999));
        $address = {$this-street}, {$this-city};

        imagestring($image, 5, 120, 100, strtoupper($this-businessName), $blue);
        imagestring($image, 5, 120, 170, $name, $black);
        imagestring($image, 4, 120, 210, $this-position, $blue);

        imageline($image, 120, 260, 880, 260, $gray);

        imagestring($image, 4, 120, 310, 'Email ' . $email, $black);
        imagestring($image, 4, 120, 365, 'Phone ' . $phone, $black);
        imagestring($image, 4, 120, 420, 'Address ' . $address, $black);
        imagestring($image, 3, 120, 485, $this-website, $gray);

        return $image;
    }
}
